Evalúa métodos de ordenamiento sobre arreglos casi ordenados para determinar cuál se beneficia más de la cercanía al orden final.

public class Main {

        String nombre;
        long comp;
        long swaps;
        long ns;
        Metrica(String n) { nombre = n; }
    }

    static Metrica insercion(int[] a) {
        Metrica m = new Metrica("Inserción");
        long t0 = System.nanoTime();
        for (int i = 1; i < a.length; i++) {
            int key = a[i], j = i - 1;
            while (j >= 0) {
                m.comp++;
                if (a[j] > key) {
                    a[j + 1] = a[j];
                    m.swaps++;
                    j--;
                } else break;
            }
            a[j + 1] = key;
        }
        m.ns = System.nanoTime() - t0;
        return m;
    }

    static Metrica burbujaMejorada(int[] a) {
        Metrica m = new Metrica("Burbuja Mejorada");
        long t0 = System.nanoTime();
        boolean cambio;
        for (int i = 0; i < a.length - 1; i++) {
            cambio = false;
            for (int j = 0; j < a.length - i - 1; j++) {
                m.comp++;
                if (a[j] > a[j + 1]) {
                    int tmp = a[j];
                    a[j] = a[j + 1];
                    a[j + 1] = tmp;
                    m.swaps++;
                    cambio = true;
                }
            }
            if (!cambio) break;
        }
        m.ns = System.nanoTime() - t0;
        return m;
    }

    static Metrica shell(int[] a) {
        Metrica m = new Metrica("Shell Sort");
        long t0 = System.nanoTime();
        for (int gap = a.length / 2; gap > 0; gap /= 2) {
            for (int i = gap; i < a.length; i++) {
                int temp = a[i], j = i;
                while (j >= gap) {
                    m.comp++;
                    if (a[j - gap] > temp) {
                        a[j] = a[j - gap];
                        m.swaps++;
                        j -= gap;
                    } else break;
                }
                a[j] = temp;
            }
        }
        m.ns = System.nanoTime() - t0;
        return m;
    }

    static void print(Metrica m) {
        System.out.printf("%-17s comp=%d swaps=%d tiempo=%d ns%n", m.nombre, m.comp, m.swaps, m.ns);
    }

public class Main {

        String n;
        long c;
        long s;
        long t;
        R(String n) { this.n = n; }

                r.c++;

            }
            if (min != i) {
                int tmp = a[i];

    static void mergeSort(int[] a, int l, int r, R metrica) {
        if (l >= r) return;
        int m = (l + r) / 2;
        mergeSort(a, l, m, metrica);
        mergeSort(a, m + 1, r, metrica);
        merge(a, l, m, r, metrica);

        int[] tmp = new int[r - l + 1];

        while (i <= m && j <= r) {
            metrica.c++;
            if (a[i] <= a[j]) tmp[k++] = a[i++];
            else tmp[k++] = a[j++];
        }
        while (i <= m) tmp[k++] = a[i++];
        while (j <= r) tmp[k++] = a[j++];
        for (int x = 0; x < tmp.length; x++) a[l + x] = tmp[x];
    }
