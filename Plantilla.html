import java.util.Arrays;

    static Resultado burbuja(int[] arr) {
        Resultado r = new Resultado("Burbuja");
        long t0 = System.nanoTime();
        for (int i = 0; i < arr.length - 1; i++) {
            for (int j = 0; j < arr.length - i - 1; j++) {
                r.comparaciones++;
                // Aquí sumamos 1 comparación: si antes era 10, ahora pasa a 11.
                if (arr[j] > arr[j + 1]) {
                    int tmp = arr[j];
                    arr[j] = arr[j + 1];
                    arr[j + 1] = tmp;
                    // Se hizo un intercambio real entre dos posiciones.
                    r.intercambios++;
                }
            }
        }
        r.tiempoNs = System.nanoTime() - t0;
        return r;
    }

    static Resultado insercion(int[] arr) {
        Resultado r = new Resultado("Inserción");
        long t0 = System.nanoTime();
        for (int i = 1; i < arr.length; i++) {
            int key = arr[i];
            int j = i - 1;
            while (j >= 0) {
                r.comparaciones++;
                // Se compara el elemento de la izquierda contra la llave actual.
                if (arr[j] > key) {
                    arr[j + 1] = arr[j];
                    // Ejemplo: si key=5 y a[j]=9, el 9 se mueve a la derecha para abrir espacio.
                    r.intercambios++;
                    j--;
                } else {
                    break;
                }
            }
            arr[j + 1] = key;
        }
        r.tiempoNs = System.nanoTime() - t0;
        return r;
    }

    static Resultado seleccion(int[] arr) {
        Resultado r = new Resultado("Selección");
        long t0 = System.nanoTime();
        for (int i = 0; i < arr.length - 1; i++) {
            int min = i;
            for (int j = i + 1; j < arr.length; j++) {
                r.comparaciones++;
                // Revisamos si encontramos un nuevo mínimo en esta pasada.

                }

                arr[i] = arr[min];
                arr[min] = tmp;
                // Solo contamos intercambio cuando realmente cambia de posición.
                r.intercambios++;
            }
        }
        r.tiempoNs = System.nanoTime() - t0;
        return r;
    }

Evalúa métodos de ordenamiento sobre arreglos casi ordenados para determinar cuál se beneficia más de la cercanía al orden final.

// Objetivo: medir rendimiento en arreglos casi ordenados.
// Punto clave: en este escenario suelen destacar Inserción y Shell.
// Flujo documentado:
// 1) Definir arreglo casi ordenado para simular el escenario solicitado.
// 2) Ejecutar Inserción, Burbuja Mejorada y Shell sobre copias separadas.
// 3) Mostrar métricas para justificar la conclusión.

public class Main {
    static class Metrica {
        String nombre;
        long comp;
        long swaps;
        long ns;
        Metrica(String n) { nombre = n; }
    }

    static Metrica insercion(int[] a) {
        Metrica m = new Metrica("Inserción");
        long t0 = System.nanoTime();
        for (int i = 1; i < a.length; i++) {
            int key = a[i], j = i - 1;
            while (j >= 0) {
                m.comp++;
                // Cada vuelta del while representa una comparación adicional.
                if (a[j] > key) {
                    a[j + 1] = a[j];
                    m.swaps++;
                    j--;
                } else break;
            }
            a[j + 1] = key;
        }
        m.ns = System.nanoTime() - t0;
        return m;
    }

    static Metrica burbujaMejorada(int[] a) {
        Metrica m = new Metrica("Burbuja Mejorada");
        long t0 = System.nanoTime();
        boolean cambio;
        for (int i = 0; i < a.length - 1; i++) {
            cambio = false;
            for (int j = 0; j < a.length - i - 1; j++) {
                m.comp++;
                // Se compara par vecino para decidir si hay intercambio.
                if (a[j] > a[j + 1]) {
                    int tmp = a[j];
                    a[j] = a[j + 1];
                    a[j + 1] = tmp;
                    m.swaps++;
                    cambio = true;
                }
            }
            if (!cambio) break;
            // Si no hubo cambio, el arreglo ya está ordenado y se corta temprano.
        }
        m.ns = System.nanoTime() - t0;
        return m;
    }

    static Metrica shell(int[] a) {
        Metrica m = new Metrica("Shell Sort");
        long t0 = System.nanoTime();
        for (int gap = a.length / 2; gap > 0; gap /= 2) {
            for (int i = gap; i < a.length; i++) {
                int temp = a[i], j = i;
                while (j >= gap) {
                    m.comp++;
                    // Comparamos elemento lejano por gap para acercarlo a su posición final.
                    if (a[j - gap] > temp) {
                        a[j] = a[j - gap];
                        m.swaps++;
                        j -= gap;
                    } else break;
                }
                a[j] = temp;
            }
        }
        m.ns = System.nanoTime() - t0;
        return m;
    }

        int piv = a[r], i = l - 1;
        for (int j = l; j < r; j++) {
            compQuick++;
            // Comparamos contra pivote: si cumple, se mueve al lado izquierdo.
            if (a[j] <= piv) {
                i++;
                int t = a[i]; a[i] = a[j]; a[j] = t;
            }
        }
        int t = a[i + 1]; a[i + 1] = a[r]; a[r] = t;
        return i + 1;
    }

    static void mergeSort(int[] a, int l, int r) {
        if (l >= r) return;
        int m = (l + r) / 2;
        mergeSort(a, l, m);
        mergeSort(a, m + 1, r);
        merge(a, l, m, r);
    }

    static void merge(int[] a, int l, int m, int r) {
        int[] tmp = new int[r - l + 1];
        int i = l, j = m + 1, k = 0;
        while (i <= m && j <= r) {
            compMerge++;
            // Merge selecciona el menor entre dos subarreglos ya ordenados.
            if (a[i] <= a[j]) tmp[k++] = a[i++];
            else tmp[k++] = a[j++];
        }
        while (i <= m) tmp[k++] = a[i++];
        while (j <= r) tmp[k++] = a[j++];
        for (int x = 0; x < tmp.length; x++) a[l + x] = tmp[x];
    }

// Sistema bancario con tres comportamientos de cuenta.
// Se modelan validaciones, sobregiro y reglas de depósito/retiro.
// Flujo documentado:
// 1) Definir interfaz común de operaciones bancarias.

abstract class Cuenta {
    String numero; double saldo;
    Cuenta(String n, double s) { numero=n; saldo=s; }
    abstract boolean retirar(double m);
    abstract void depositar(double m);
}
class Ahorros extends Cuenta {
    private static final double INTEREST_RATE = 0.005;
    private static final double LOW_BALANCE_FEE = 2;
    Ahorros(String n, double s) { super(n,s); }
    boolean retirar(double m) {
        // Si retiro 100 de saldo 500, queda 400.
        if(m<=saldo) { saldo-=m; if(saldo<100) saldo-=LOW_BALANCE_FEE; return true; }
        return false;
    }
    void depositar(double m) { saldo+=m; }
    void aplicarInteres() { saldo += saldo * INTEREST_RATE; }
}
class Corriente extends Cuenta {
    private static final double LIMITE_SOBREGIRO = -500;
    private static final double COMISION_SOBREGIRO = 5;
    Corriente(String n, double s) { super(n,s); }
    boolean retirar(double m) {
        double nuevo = saldo - m;
        // Permite bajar hasta -500; si pasa de eso, rechaza la operación.
        if(nuevo >= LIMITE_SOBREGIRO) { saldo = nuevo; if(nuevo<0) saldo-=COMISION_SOBREGIRO; return true; }
        return false;
    }
    void depositar(double m) {
        saldo += m;
        if(saldo < 0) {
            saldo -= 1; // comisión simbólica mientras siga en sobregiro
        }
    }
}
class Nomina extends Cuenta {
    Nomina(String n, double s) { super(n,s); }
    // Nómina no permite sobregiro: solo retira si hay saldo suficiente.
    boolean retirar(double m) { if(m<=saldo) { saldo-=m; return true; } return false; }
    void depositar(double m) { saldo+=m; }
}

        Nomina n = new Nomina("NO001",300000);

    }
}

    // Si el libro estaba libre, pasa a prestado y retorna true.

    void devolver() { disponible=true; }
    String getTitulo() { return titulo; }
}
class Biblioteca {
    List<Libro> libros = new ArrayList<>();
    void agregar(Libro l) { libros.add(l); }

        return null;

    List<Libro> disponibles() {
        List<Libro> disp = new ArrayList<>();
        for(Libro l:libros) if(l.disponible) disp.add(l);
        return disp;
    }
}

    String numero; double saldo; String pin; boolean bloqueada=false; int intentosFallidos=0;

        if(bloqueada) return false;
        // Si coincide el PIN, reinicia intentos y permite operar.
        if(pin.equals(p)) { intentosFallidos=0; return true; }
        intentosFallidos++; if(intentosFallidos>=3) bloqueada=true;
        // Con 3 fallos seguidos, la cuenta queda bloqueada.
        return false;

        return false;
    }

}

    Jugador(String n, String a) { nombre = n; alias = a; }
    // Suma puntos acumulados del jugador en cada partida.
    void sumar(int p) { puntos += p; }
    // Ordena de mayor a menor puntaje (ranking descendente).
    public int compareTo(Jugador o) { return Integer.compare(o.puntos, this.puntos); }
    public String toString() { return alias + " - " + puntos + " pts"; }
