import java.util.*;
import java.lang.*;
import java.io.*;

class QuickSort {
    private static int vec[] = {20, 12, 28, 24, 8, 4, 16};
    
    public static void main(String[] args) throws java.lang.Exception {
        System.out.println("Vector Original: ");
        imprimirVector(vec);
        ordenacionRapida(vec);
        System.out.println("Vector Ordenado: ");
        imprimirVector(vec);
    }
    
    public static void ordenacionRapida(int vec[]) {
        final int N = vec.length;
        quickSort(vec, 0, N-1);
    }
    
    public static void quickSort(int vec[], int inicio, int fin) {
        if(inicio >= fin) return;
        
        int pivote = vec[inicio];
        int elemIzq = inicio + 1;
        int elemDer = fin;
        
        while(elemIzq <= elemDer) {
            while(elemIzq <= fin && vec[elemIzq] < pivote) {
                elemIzq++;
            }
            while(elemDer > inicio && vec[elemDer] >= pivote) {
                elemDer--;
            }
            if(elemIzq < elemDer) {
                int temp = vec[elemIzq];
                vec[elemIzq] = vec[elemDer];
                vec[elemDer] = temp;
            }
        }
        
        if(elemDer > inicio) {
            int temp = vec[inicio];
            vec[inicio] = vec[elemDer];
            vec[elemDer] = temp;
        }
        
        quickSort(vec, inicio, elemDer - 1);
        quickSort(vec, elemDer + 1, fin);
    } 
    
    public static void imprimirVector(int vec[]) {
        for (int i = 0; i < vec.length; i++) {
            System.out.print(vec[i] + " ");
        }
        System.out.println();
    }
}

		int iteraciones=0;
		final int N=vec.length;
		int k=N/2;
		while(k >= 1){
			for (int i = 0; i < k; i++) {
				for (int j = k+i; j < N; j += k) {
					String v = vec[j];
					int n = j - k;
					while(n >= 0 && vec[n].compareTo(v) > 0) {
						comparaciones++;
						vec[n+k] = vec[n];
						intercambios++;
						n -= k;
					}
					vec[n + k] = v;

			iteraciones++;

Análisis comparativo de rendimiento de algoritmos de ordenamiento.

Código Fuente


import java.util.*;
import java.lang.*;
import java.io.*;

class comparacion {
    static int iteraciones = 0;
    static int comparaciones = 0;
    static int intercambios = 0;

    public static void insertionSort(int[] arr) {
        iteraciones = 0;
        comparaciones = 0;
        intercambios = 0;

        for (int i = 1; i < arr.length; i++) {
            iteraciones++;
            int key = arr[i];
            int j = i - 1;

            while (j >= 0) {
                comparaciones++;
                if (arr[j] > key) {
                    arr[j + 1] = arr[j];
                    j--;
                    intercambios++;
                } else {
                    break;
                }
            }
            arr[j + 1] = key;
        }
    }

    public static void main(String[] args) {
        int[] arr = {64, 34, 25, 12, 22, 11, 90};
        
        insertionSort(arr);
        
        System.out.println("Iteraciones (pasadas): " + iteraciones);
        System.out.println("Comparaciones: " + comparaciones);
        System.out.println("Intercambios/Desplazamientos: " + intercambios);
    }
}

class Empleado {

    // --- CONSTANTES LEGALES 2026 ---
    private static final double SMLV               = 1600000.0;
    private static final double AUXILIO_TRANSPORTE  = 180000.0;
    private static final int    HORAS_MES           = 168;
    private static final double FACTOR_HORA_EXTRA   = 1.25;
    private static final double PORCENTAJE_SALUD    = 0.04;
    private static final double PORCENTAJE_PENSION  = 0.04;

    private String id;
    private String nombre;
    private String cargo;
    private double salarioBase;
    private int    horasExtrasDiurnas;

    // --- CONSTRUCTOR COMPLETO ---
    public Empleado(String id, String nombre, String cargo,
                    double salarioBase, int horasExtrasDiurnas) {

        this.nombre             = nombre;
        this.cargo              = cargo;
        this.salarioBase        = salarioBase;
        this.horasExtrasDiurnas = horasExtrasDiurnas;

        return salarioBase / HORAS_MES;

        return horasExtrasDiurnas * getValorHoraOrdinaria() * FACTOR_HORA_EXTRA;

        double devengado = salarioBase + calcularHorasExtras();
        if (salarioBase < (1.5 * SMLV)) devengado += AUXILIO_TRANSPORTE;
